```python
import math

def is_prime(n):
    if n < 2:
        return False
    for i in range(2, int(math.sqrt(n)) + 1):
        if n % i == 0:
            return False
    return True

primes = [n for n in range(2, 100001) if is_prime(n)]

print("| Number | Prime |")
print("| --- | --- |")
for prime in primes:
    print(f"| {prime} | {'Yes' if is_prime(prime) else 'No'} |")
```
